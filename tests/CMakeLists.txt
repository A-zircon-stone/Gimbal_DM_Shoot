# 直接提取原温控函数，不重新实现状态机；无需STM32硬件即可回归两处反写。
set(INS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../User/Tasks/INS_Task/INSTask.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${INS_SOURCE}")
file(READ "${INS_SOURCE}" INS_TEXT)
string(FIND "${INS_TEXT}" "void IMU_Temperature_Ctrl(void)" TEMP_BEGIN)
string(FIND "${INS_TEXT}" "//------------------------------------functions below" TEMP_END)
if(TEMP_BEGIN LESS 0 OR TEMP_END LESS TEMP_BEGIN)
    message(FATAL_ERROR "未找到旧温控函数的测试边界")
endif()
math(EXPR TEMP_LENGTH "${TEMP_END}-${TEMP_BEGIN}")
string(SUBSTRING "${INS_TEXT}" ${TEMP_BEGIN} ${TEMP_LENGTH} TEMP_FUNCTIONS)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/legacy_temperature.inc" "${TEMP_FUNCTIONS}")
add_executable(imu_temperature_legacy_test imu_temperature_legacy_test.cpp)
target_compile_features(imu_temperature_legacy_test PRIVATE cxx_std_17)
target_include_directories(imu_temperature_legacy_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME legacy_temperature_cold COMMAND imu_temperature_legacy_test cold)
add_test(NAME legacy_temperature_hot COMMAND imu_temperature_legacy_test hot)
add_test(NAME legacy_temperature_pid_order COMMAND imu_temperature_legacy_test pid)
